**When using VSCode if your header files do not line in with #include "base.h"
where base.h is not in code but is in code/common then to satisfy VSCode intellisense
update the c_cpp_properties.json
#+BEGIN_SRC json
{
    "configurations": [
        {
            "name": "Win32",
            "includePath": [
                "${workspaceFolder}/**",
                "${workspaceFolder}/wing/code/common",
                "${VULKAN_SDK}\\Include",
                "C:\\VulkanSDK\\1.4.341.1\\Include"
            ],
            "defines": [
                "_DEBUG",
                "UNICODE",
                "_UNICODE"
            ],
            "windowsSdkVersion": "10.0.22621.0",
            "compilerPath": "cl.exe",
            "cStandard": "c17",
            "cppStandard": "c++17",
            "intelliSenseMode": "windows-msvc-x64"
        }
    ],
    "version": 4
}
#+END_SRC

then do ctrl+shift+p and restart intellisense or restart intellisense server / active file
