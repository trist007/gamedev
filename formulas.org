Vector3f normalizeFloat3(Vector3f vector) {
  float length = sqrtf(vector.x*vector.x + vector.y*vector.y + vector.z*vector.z);

  vector.x = vector.x / length;
  vector.y = vector.y / length;
  vector.z = vector.z / length;

  return vector;
}
